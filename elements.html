Data Cleaning Automation (Python)

import pandas as pd

def clean_student_data(input_file, output_file):
    # Load raw data
    df = pd.read_csv(input_file)
    
    # Drop missing values and reset index
    cleaned_df = df.dropna().reset_index(drop=True)
    
    # Save as cleaned_student_data.csv
    cleaned_df.to_csv(output_file, index=False)
    print(f"Data successfully cleaned and saved to {output_file}")

clean_student_data('raw_data.csv', 'cleaned_student_.csv')
